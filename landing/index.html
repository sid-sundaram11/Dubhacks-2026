LOCAL RUNBOOK BASH

# 1. Install dependencies

$ npm install


# 2. Start SAM3 segmentation service (Python, port 8790)

$ cd services/sam3 && source .venv/bin/activate && python app.py


# 3. Start geometry & landing server (port 8787)

$ npm run server


# 4. Launch Expo mobile starfield

$ npx expo start
